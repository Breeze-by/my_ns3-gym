cmake_minimum_required(VERSION 3.10)
project(mr3_smoother_clock_probe LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav2_velocity_smoother REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(lifecycle_msgs REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(smoother_clock_probe smoother_clock_probe.cpp)
ament_target_dependencies(smoother_clock_probe rclcpp nav2_velocity_smoother
  rosgraph_msgs geometry_msgs lifecycle_msgs)
target_link_libraries(smoother_clock_probe ${CMAKE_DL_LIBS})
target_compile_options(smoother_clock_probe PRIVATE -Wall -Wextra -Wpedantic)
