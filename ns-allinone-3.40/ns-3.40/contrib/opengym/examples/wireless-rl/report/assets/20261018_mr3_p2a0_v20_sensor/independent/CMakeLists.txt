cmake_minimum_required(VERSION 3.16)
project(mr3_startup_sensor_recorder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/mr3_startup_sensor_recorder.cpp" RECORDER_SOURCE_SHA)
add_executable(mr3_startup_sensor_recorder mr3_startup_sensor_recorder.cpp)
target_compile_options(mr3_startup_sensor_recorder PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(mr3_startup_sensor_recorder PRIVATE
  MR3_RECORDER_SOURCE_SHA256="${RECORDER_SOURCE_SHA}")
ament_target_dependencies(mr3_startup_sensor_recorder
  rclcpp sensor_msgs nav_msgs geometry_msgs tf2_msgs rosgraph_msgs rcl_interfaces)
target_link_libraries(mr3_startup_sensor_recorder nlohmann_json::nlohmann_json OpenSSL::Crypto)
