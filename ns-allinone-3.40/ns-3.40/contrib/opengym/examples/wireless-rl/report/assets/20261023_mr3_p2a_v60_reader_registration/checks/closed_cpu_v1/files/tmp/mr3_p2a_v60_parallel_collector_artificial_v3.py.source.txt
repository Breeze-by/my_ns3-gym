"""Four regular files + one external literal link; no activity/raw collector run."""
import gzip
import hashlib
import importlib.util
import json
from pathlib import Path
import tempfile

SOURCE = Path('/tmp/mr3_p2a_v60_collect_closed_draft_v3.py')
spec = importlib.util.spec_from_file_location('_only_artificial_parallel_collector', SOURCE)
module = importlib.util.module_from_spec(spec)
spec.loader.exec_module(module)


def digest(path):
    raw = path.read_bytes()
    return hashlib.sha256(raw).hexdigest(), len(raw)


def main():
    with tempfile.TemporaryDirectory(prefix='mr3-v60-parallel-artificial-') as temporary:
        root = Path(temporary)
        inputs = root / 'raw'
        inputs.mkdir()
        payloads = {
            'a.jsonl': b'{"fixture":true,"zero":0}\n' * 16,
            'b.bin': bytes(range(256)) * 4,
            'c.csv': b'a,b\n0,1\n' * 32,
            'd.gz': gzip.compress(b'opaque-existing-gzip-fixture\n', mtime=0),
        }
        for name, raw in payloads.items():
            (inputs / name).write_bytes(raw)
        (inputs / 'external-link').symlink_to('/nonexistent-external-fixture-must-not-be-followed')
        # Patch every discovery root before discovery; real /tmp activity roots are never enumerated.
        module.PREFIX = root.name + '/raw'
        module.NATIVE, module.POLICY, module.ROOT_TOOL = [
            root / name for name in ('missing-native', 'missing-policy', 'missing-root-tool')]
        copier = module.helper()  # Loads the unchanged byte copier source only.
        baseline_source_sha = {name: digest(inputs / name) for name in payloads}

        def selected_for(name):
            module.A = root / name
            module.DEST = module.A / 'closed_execution'
            module.A.mkdir()
            copier.A, copier.DEST = module.A, module.DEST
            selected, links, directories, missing = module.discover(copier, [])
            assert len(selected) == 4 and len(links) == 1
            link = next(iter(links.values()))
            assert link['target_followed'] is False
            assert link['target_text'] == '/nonexistent-external-fixture-must-not-be-followed'
            module.DEST.mkdir()
            return selected, links

        serial, serial_links = selected_for('serial')
        serial_entries = [copier.archive(source, row) for source, row in sorted(serial.items())]
        serial_bytes = {source.name: row['target'].read_bytes() for source, row in serial.items()}
        parallel, parallel_links = selected_for('parallel')
        entries, attempts, errors = module.archive_regulars(copier, parallel)
        assert not errors and len(entries) == len(attempts) == 4
        assert [row['original'] for row in entries] == [str(path) for path in sorted(parallel)]
        for source, row in parallel.items():
            assert row['target'].read_bytes() == serial_bytes[source.name]
        for entry in entries:
            assert entry['source_stat_stable'] and entry['decoded_bytes_verified']
            source = Path(entry['original'])
            target = module.A / entry['canonical']
            decoded = gzip.open(target, 'rb').read() if entry['codec'] == 'gzip-mtime0-lossless' else target.read_bytes()
            assert decoded == payloads[source.name]
        assert all(digest(inputs / name) == proof for name, proof in baseline_source_sha.items())
        assert next(iter(serial_links.values()))['link_stat'] == next(iter(parallel_links.values()))['link_stat']

        failure, _ = selected_for('failure')
        original_archive = copier.archive
        def fail_one(source, row):
            if source.name == 'a.jsonl':
                row['target'].parent.mkdir(parents=True, exist_ok=True)
                row['target'].write_bytes(b'preserved-artificial-partial')
                raise ValueError('injected artificial single-file copy failure')
            return original_archive(source, row)
        copier.archive = fail_one
        completed, attempted, failed = module.archive_regulars(copier, failure)
        assert len(completed) == 3 and len(attempted) == 4 and len(failed) == 1
        assert failed[0]['original'].endswith('/a.jsonl')
        assert Path(failed[0]['target']).read_bytes() == b'preserved-artificial-partial'
        assert [row['original'] for row in completed] == sorted(row['original'] for row in completed)
        assert all(digest(inputs / name) == proof for name, proof in baseline_source_sha.items())
    print(json.dumps({'passed': True, 'artificial_regular_files': 4, 'literal_links': 1,
        'serial_parallel_canonical_bytes_identical': True, 'decoded_original_bytes_verified': True,
        'failure_all_submitted_jobs_joined': True, 'failure_three_completed_and_one_partial_preserved': True,
        'actual_activity_inputs_read': False, 'real_collector_main_called': False}))


if __name__ == '__main__':
    main()
