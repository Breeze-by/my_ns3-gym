from pathlib import Path

R = Path('/home/zhuyulab/ns3-workspace')
W = R / 'ns-allinone-3.40/ns-3.40/contrib/opengym/examples/wireless-rl'
A = W / 'report/assets/20261024_mr3_p2a_v61_cancel_forensics'
OUT = Path('/tmp/mr3_p2a_v61_artificial_cpu_v2')

                A / 'scope_v1.json', A / 'extraction_spec_v1.json',
                Path(scope['entry_ledger']['path']), Path(spec['closed_inventory']['path']),
                Path(spec['closed_quiescence_review']['path'])))

        '--scope', 'cpu', '--output', str(OUT)]
for source in sorted(sources):
    argv += ['--source', str(source)]
argv += ['--', '/usr/bin/python3', '-B', '-m', 'pytest', '-p', 'no:cacheprovider',
         '-q', str(A / 'test_cancel_forensics_v1.py'), '--junitxml=' + str(OUT / 'junit.xml')]
os.execv(argv[0], argv)
